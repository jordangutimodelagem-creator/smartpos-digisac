async function searchProduct() { const q = document.getElementById('searchInput').value; if (q.length < 2) return; try { const res = await fetch(`${API_URL}/produtos?search=${encodeURIComponent(q)}`); const data = await res.json(); // Garante que 'products' é sempre um array products = Array.isArray(data) ? data : []; const select = document.getElementById('productSelect'); select.innerHTML = ''; if (products.length === 0) { const opt = document.createElement('option'); opt.textContent = 'Nenhum produto encontrado'; select.appendChild(opt); return; } products.forEach(p => { const opt = document.createElement('option'); opt.value = p.id || p._id; opt.textContent = `${p.name || p.title} - R$ ${p.price || p.sale_price || 0}`; select.appendChild(opt); }); } catch (e) { console.error('Erro ao buscar produtos:', e); } }
